    14→def parse_num(s):
    15→    """Parse a number string like '1,234,567' or '-' or '' to int."""
    16→    if s is None:
    17→        return None
    18→    s = str(s).strip()
    19→    # Handle negative in parentheses
    20→    m = re.match(r'^\(([\d,]+)\)$', s)
    21→    if m:
    22→        return -int(m.group(1).replace(',', ''))
    23→    s = s.replace(',', '')
    24→    if s in ('', '-', '0', '00', '( )'):
    25→        return 0
    26→    try:
    27→        return int(s)
    28→    except ValueError:
    29→        return None
