   752→      const sheet = workbook.Sheets[sheetName]
   753→      const rows = XLSX.utils.sheet_to_json<Record<string, unknown>>(sheet, { defval: '' })
   754→      const parsedEntries: TBEntryUI[] = []
   755→      for (let i = 0; i < rows.length; i++) {
   756→        const row = rows[i]
   757→        const values = Object.values(row)
   758→        if (values.every(v => v === '' || v === 0 || v === undefined || v === null)) continue
   759→        const code = String(row['Account Code'] ?? row['Code'] ?? row['Acct Code'] ?? values[0] ?? '').trim()
   760→        const name = String(row['Account Name'] ?? row['Description'] ?? row['Account'] ?? values[1] ?? '').trim()

   762→        const classification = String(row['Classification'] ?? row['Type'] ?? values[2] ?? 'revenue-non-exchange').trim()
   763→        const category = String(row['Category'] ?? row['Line Item'] ?? values[3] ?? '').trim()
   764→        const noteRef = String(row['Note Ref'] ?? row['Note'] ?? values[4] ?? '').trim()
   765→        const parseNum = (v: unknown): number => {
   766→          if (typeof v === 'number') return v
   767→          if (typeof v === 'string') { const c = v.replace(/[^\d.-]/g, ''); const n = parseFloat(c); return isNaN(n) ? 0 : n }
   768→          return 0
   769→        }

   771→          accountCode: code, accountName: name, classification, category, noteRef,
   772→          debitCurrent: parseNum(row['Dr Current'] ?? row['Debit Current'] ?? row['Debit'] ?? values[5]),
   773→          creditCurrent: parseNum(row['Cr Current'] ?? row['Credit Current'] ?? row['Credit'] ?? values[6]),
   774→          debitPrior: parseNum(row['Dr Prior'] ?? row['Debit Prior'] ?? values[7]),
   775→          creditPrior: parseNum(row['Cr Prior'] ?? row['Credit Prior'] ?? values[8]),
   776→          budgetInitial: parseNum(row['Budget Initial'] ?? row['Initial Budget'] ?? values[9]),
   777→          budgetAdjusted: parseNum(row['Budget Adj'] ?? row['Adjusted Budget'] ?? values[10]),
   778→          _id: Math.random().toString(36).substring(2, 9),
   779→        })
   780→      }
   781→      if (parsedEntries.length === 0) throw new Error('No valid entries found in Excel file')
   782→      set({ entries: parsedEntries, isLoading: false })
   783→      toast.success(`Imported ${parsedEntries.length} entries from Excel`)
   784→    } catch (err) {
   785→      set({ isLoading: false })
   786→      const msg = err instanceof Error ? err.message : 'Failed to parse Excel file'

   830→    let order = 0

   835→      const adjusted = catEntries.reduce((s, e) => s + (e.budgetAdjusted || 0), 0)

   837→      const actual = catEntries.reduce((s, e) => {
   838→        const cls = e.classification
   839→        return s + (cls === 'revenue-non-exchange' || cls === 'revenue-exchange' ? e.creditCurrent : e.debitCurrent)
   840→      }, 0)
   841→      const variance = revised - actual
   842→      const variancePercent = revised !== 0 ? (variance / revised) * 100 : 0
   843→      budgetEntries.push({
   844→        category: cat.key, categoryLabel: cat.label, lineItem: '',
   845→        initialBudget: initial, revisedBudget: revised, virementFrom: 0, virementTo: adjusted,
   846→        actualAmount: actual, variance, variancePercent, sortOrder: order++,
   847→      })
   848→    }
