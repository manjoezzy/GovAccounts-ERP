   165→function WizardProgressBar() {
   166→  const { wizardStep, setWizardStep, entries, supplementaryData, complianceData, generatedStatements } = useFinancialStore()
   167→
   168→  const stepComplete = (s: number): boolean => {
   169→    if (s === 1) return true
   170→    if (s === 2) return entries.length > 0
   171→    if (s === 3) return true // budget is optional
   172→    if (s === 4) return supplementaryData !== null
   173→    if (s === 5) return supplementaryData !== null
   174→    if (s === 6) return complianceData !== null || generatedStatements !== null
   175→    return false
   176→  }
   177→
   178→  return (
   179→    <div className="bg-muted/50 border-b border-border">
   180→      <div className="max-w-[1600px] mx-auto px-4 sm:px-6">
   181→        <div className="flex items-center gap-1 py-2 overflow-x-auto">
   182→          {WIZARD_STEPS.map(({ step, label, icon }) => {
   183→            const isActive = wizardStep === step
   184→            const isComplete = step < wizardStep || stepComplete(step)
   185→            const canNavigate = isComplete && step !== wizardStep
   186→            return (
   187→              <React.Fragment key={step}>
   188→                {step > 1 && <ChevronRight className="h-4 w-4 text-muted-foreground/50 flex-shrink-0" />}
   189→                <button
   190→                  onClick={() => canNavigate ? setWizardStep(step as WizardStep) : undefined}
   191→                  disabled={!canNavigate && !isActive}
   192→                  className={`flex items-center gap-2 px-3 py-1.5 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${
   193→                    isActive
   194→                      ? 'bg-primary text-primary-foreground shadow-sm'
   195→                      : isComplete
   196→                        ? 'bg-primary/10 text-primary hover:bg-primary/20 cursor-pointer'
   197→                        : 'text-muted-foreground cursor-not-allowed'
   198→                  }`}
   199→                >
   200→                  <span className="flex items-center justify-center h-5 w-5 rounded-full text-[10px] font-bold border current:border-current">
   201→                    {isComplete && !isActive ? <CheckCircle2 className="h-3.5 w-3.5" /> : icon}
   202→                  </span>
   203→                  <span className="hidden sm:inline">{label}</span>
   204→                </button>
   205→              </React.Fragment>
   206→            )
   207→          })}
   208→        </div>
   209→      </div>
   210→    </div>
   211→  )
   212→}

   218→function StatusBar() {
   219→  const { entries, supplementaryData, complianceData, approvalWorkflow } = useFinancialStore()
   220→  const totalDr = entries.reduce((s, e) => s + (e.debitCurrent || 0), 0)
   221→  const totalCr = entries.reduce((s, e) => s + (e.creditCurrent || 0), 0)
   222→  const tbBalanced = Math.abs(totalDr - totalCr) < 0.01
   223→
   224→  const suppFields = supplementaryData
   225→    ? ['accountingOfficer', 'chiefFinanceOfficer', 'internalAuditHead', 'depreciationRate', 'employeeCount']
   226→        .filter(k => { const v = supplementaryData[k as keyof typeof supplementaryData]; return v !== '' && v !== 0 })
   227→    : []
   228→  const suppComplete = suppFields.length
   229→
   230→  const approvalStep = approvalWorkflow?.currentStep || 'draft'
   231→
   232→  return (
   233→    <div className="flex flex-wrap items-center gap-x-4 gap-y-1 px-4 sm:px-6 py-2 text-[11px] text-muted-foreground bg-card border-b border-border">
   234→      <span className="flex items-center gap-1">
   235→        Trial Balance: {tbBalanced ? <CheckCircle2 className="h-3 w-3 text-primary" /> : <X className="h-3 w-3 text-destructive" />}
   236→        <span className={tbBalanced ? 'text-primary font-medium' : 'text-destructive font-medium'}>{tbBalanced ? 'Balanced' : 'Unbalanced'}</span>
   237→      </span>
   238→      <span className="flex items-center gap-1">
   239→        Supplementary: <span className="font-medium text-foreground">{suppComplete}/5</span>
   240→      </span>
   241→      <span className="flex items-center gap-1">
   242→        Validation: {complianceData ? <span className="font-medium text-primary">{complianceData.score}% compliant</span> : <span className="font-medium">Not run</span>}
   243→      </span>
   244→      <span className="flex items-center gap-1">

   343→function Step2TrialBalance() {
   344→  const {
   345→    entries, entityConfig, setEntityConfig, updateEntry, deleteEntry, addEntry,
   346→    loadTemplate, saveTrialBalance, isLoading, wizardStep, setWizardStep, autoBalance,
   347→    clearTrialBalance, uploadExcelFile,
   348→  } = useFinancialStore()
   349→  const [autoBalancing, setAutoBalancing] = useState(false)
   350→  const [showAutoBalanceDialog, setShowAutoBalanceDialog] = useState(false)
   351→  const [showClearConfirm, setShowClearConfirm] = useState(false)
   352→  const fileInputRef = React.useRef<HTMLInputElement>(null)
   353→
   354→  const totalDebitCurrent = entries.reduce((s, e) => s + (e.debitCurrent || 0), 0)
   355→  const totalCreditCurrent = entries.reduce((s, e) => s + (e.creditCurrent || 0), 0)
   356→  const totalDebitPrior = entries.reduce((s, e) => s + (e.debitPrior || 0), 0)
   357→  const totalCreditPrior = entries.reduce((s, e) => s + (e.creditPrior || 0), 0)
   358→  const diffCurrent = totalDebitCurrent - totalCreditCurrent
   359→  const diffPrior = totalDebitPrior - totalCreditPrior
   360→  const isBalancedCurrent = Math.abs(diffCurrent) < 0.01
   361→  const isBalancedPrior = Math.abs(diffPrior) < 0.01
   362→  const isBalanced = isBalancedCurrent && isBalancedPrior
   363→
   364→  const handleSave = async () => {
   365→    try {
   366→      await saveTrialBalance()
   367→      toast.success('Trial balance saved and validated')
   368→    } catch {
   369→      toast.error('Failed to save trial balance')
   370→    }
   371→  }
   372→
   373→  const handleAutoBalance = async () => {
   374→    setAutoBalancing(true)
   375→    try {
   376→      await autoBalance()
   377→      toast.success('Trial balance auto-balanced')
   378→      setShowAutoBalanceDialog(false)
   379→    } catch {
   380→      toast.error('Failed to auto-balance')
   381→    } finally {
   382→      setAutoBalancing(false)
   383→    }
   384→  }
   385→
   386→  return (
   387→    <motion.div {...pageVariants} className="py-4 space-y-4">
   388→      {/* Balance Indicator Banner */}
   389→      <Card className={`border-l-4 ${isBalanced ? 'border-l-primary bg-primary/5' : 'border-l-destructive bg-destructive/5'}`}>
   390→        <CardContent className="py-3">
   391→          <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   392→            <div className="flex items-center gap-3">
   393→              {isBalanced
   394→                ? <CheckCircle2 className="h-5 w-5 text-primary" />
   395→                : <AlertCircle className="h-5 w-5 text-destructive" />}
   396→              <div>
   397→                <p className="text-sm font-semibold text-foreground">
   398→                  {isBalanced ? 'Trial Balance is Balanced' : 'Trial Balance is NOT Balanced'}
   399→                </p>
   400→                {!isBalanced && (
   401→                  <p className="text-xs text-muted-foreground mt-0.5">
   402→                    Current diff: {entityConfig.currency} {formatNum(diffCurrent)}
   403→                    {diffPrior !== 0 && ` | Prior diff: ${entityConfig.currency} ${formatNum(diffPrior)}`}
   404→                  </p>
   405→                )}
   406→              </div>
   407→            </div>
   408→            <div className="flex items-center gap-2">
   409→              {!isBalanced && (
   410→                <Dialog open={showAutoBalanceDialog} onOpenChange={setShowAutoBalanceDialog}>
   411→                  <DialogTrigger asChild>
   412→                    <Button variant="outline" size="sm" className="gap-1.5" disabled={entries.length === 0}>
   413→                      <Wand2 className="h-4 w-4" /> Auto-Balance
   414→                    </Button>
   415→                  </DialogTrigger>
   416→                  <DialogContent>
   417→                    <DialogHeader>
   418→                      <DialogTitle>Auto-Balance Trial Balance</DialogTitle>
   419→                      <DialogDescription>
   420→                        This will create a &quot;Suspense&quot; entry to zero out the difference between debits and credits.
   421→                        Current difference: {entityConfig.currency} {formatNum(diffCurrent)}
   422→                      </DialogDescription>
   423→                    </DialogHeader>
   424→                    <DialogFooter>
   425→                      <Button variant="outline" onClick={() => setShowAutoBalanceDialog(false)}>Cancel</Button>
   426→                      <Button onClick={handleAutoBalance} disabled={autoBalancing} className="bg-primary text-primary-foreground gap-2">
   427→                        {autoBalancing ? <Loader2 className="h-4 w-4 animate-spin" /> : <Wand2 className="h-4 w-4" />}
   428→                        Auto-Balance
   429→                      </Button>
   430→                    </DialogFooter>
   431→                  </DialogContent>
   432→                </Dialog>
   433→              )}
   434→            </div>
   435→          </div>
   436→        </CardContent>
   437→      </Card>
   438→
   439→      {/* Toolbar */}
   440→      <div className="flex flex-wrap items-center gap-2">
   441→        <TooltipProvider>
   442→          <Tooltip>
   443→            <TooltipTrigger asChild>
   444→              <Button variant="outline" size="sm" className="gap-1.5" onClick={() => { loadTemplate(); toast.success('Template loaded') }}>
   445→                <FileSpreadsheet className="h-4 w-4" /> Load Template
   446→              </Button>
   447→            </TooltipTrigger>
   448→            <TooltipContent><p>Load the standard IPSAS chart of accounts template</p></TooltipContent>
   449→          </Tooltip>
   450→        </TooltipProvider>
   451→        <TooltipProvider>
   452→          <Tooltip>
   453→            <TooltipTrigger asChild>
   454→              <Button variant="outline" size="sm" className="gap-1.5" onClick={() => fileInputRef.current?.click()} disabled={isLoading}>
   455→                <Upload className="h-4 w-4" /> Upload Excel
   456→              </Button>
   457→            </TooltipTrigger>
   458→            <TooltipContent><p>Upload a trial balance from Excel (.xlsx, .xls, .csv)</p></TooltipContent>
   459→          </Tooltip>
   460→        </TooltipProvider>
   461→        <input
   462→          ref={fileInputRef}
   463→          type="file"
   464→          accept=".xlsx,.xls,.csv"
   465→          className="hidden"
