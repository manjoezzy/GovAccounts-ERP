   176→function WizardProgressBar() {
   177→  const { wizardStep, setWizardStep, entries, supplementaryData, complianceData, generatedStatements } = useFinancialStore()
   178→
   179→  const stepComplete = (s: number): boolean => {
   180→    if (s === 1) return true
   181→    if (s === 2) return true // transactions optional
   182→    if (s === 3) return true // cashbook optional
   183→    if (s === 4) return true // ledger optional
   184→    if (s === 5) return entries.length > 0
   185→    if (s === 6) return true // votebook optional
   186→    if (s === 7) return true // abstract optional
   187→    if (s === 8) return true // budget is optional
   188→    if (s === 9) return supplementaryData !== null
   189→    if (s === 10) return supplementaryData !== null
   190→    if (s === 11) return complianceData !== null || generatedStatements !== null
   191→    return false
   192→  }
   193→
   194→  return (
   195→    <div className="bg-muted/50 border-b border-border">
   196→      <div className="max-w-[1600px] mx-auto px-4 sm:px-6">
   197→        <div className="flex items-center gap-1 py-2 overflow-x-auto">
   198→          {WIZARD_STEPS.map(({ step, label, icon }) => {
   199→            const isActive = wizardStep === step
   200→            const isComplete = step < wizardStep || stepComplete(step)
   201→            const canNavigate = isComplete && step !== wizardStep
   202→            return (
   203→              <React.Fragment key={step}>
   204→                {step > 1 && <ChevronRight className="h-4 w-4 text-muted-foreground/50 flex-shrink-0" />}
   205→                <button
   206→                  onClick={() => canNavigate ? setWizardStep(step as WizardStep) : undefined}
   207→                  disabled={!canNavigate && !isActive}
   208→                  className={`flex items-center gap-2 px-3 py-1.5 rounded-lg text-xs font-medium whitespace-nowrap transition-all ${
   209→                    isActive
   210→                      ? 'bg-primary text-primary-foreground shadow-sm'
   211→                      : isComplete
   212→                        ? 'bg-primary/10 text-primary hover:bg-primary/20 cursor-pointer'
   213→                        : 'text-muted-foreground cursor-not-allowed'
   214→                  }`}
   215→                >
   216→                  <span className="flex items-center justify-center h-5 w-5 rounded-full text-[10px] font-bold border current:border-current">
   217→                    {isComplete && !isActive ? <CheckCircle2 className="h-3.5 w-3.5" /> : icon}
   218→                  </span>
   219→                  <span className="hidden sm:inline">{label}</span>
   220→                </button>
   221→              </React.Fragment>
   222→            )
   223→          })}
   224→        </div>
   225→      </div>
   226→    </div>
   227→  )
   228→}

   234→function StatusBar() {
   235→  const { entries, supplementaryData, complianceData, approvalWorkflow } = useFinancialStore()
   236→  const totalDr = entries.reduce((s, e) => s + (e.debitCurrent || 0), 0)
   237→  const totalCr = entries.reduce((s, e) => s + (e.creditCurrent || 0), 0)
   238→  const tbBalanced = Math.abs(totalDr - totalCr) < 0.01
   239→
   240→  const suppFields = supplementaryData
   241→    ? ['accountingOfficer', 'chiefFinanceOfficer', 'internalAuditHead', 'depreciationRate', 'employeeCount']
   242→        .filter(k => { const v = supplementaryData[k as keyof typeof supplementaryData]; return v !== '' && v !== 0 })
   243→    : []
   244→  const suppComplete = suppFields.length
   245→
   246→  const moduleCounts = `${transactions.length} txns · ${cashbookEntries.length} cb · ${ledgerEntries.length} lg · ${votebookEntries.length} votes`
   247→
   248→  const approvalStep = approvalWorkflow?.currentStep || 'draft'
   249→
   250→  return (
   251→    <div className="flex flex-wrap items-center gap-x-4 gap-y-1 px-4 sm:px-6 py-2 text-[11px] text-muted-foreground bg-card border-b border-border">
   252→      <span className="flex items-center gap-1">
   253→        Trial Balance: {tbBalanced ? <CheckCircle2 className="h-3 w-3 text-primary" /> : <X className="h-3 w-3 text-destructive" />}
   254→        <span className={tbBalanced ? 'text-primary font-medium' : 'text-destructive font-medium'}>{tbBalanced ? 'Balanced' : 'Unbalanced'}</span>
   255→      </span>
   256→      <span className="flex items-center gap-1">
   257→        Supplementary: <span className="font-medium text-foreground">{suppComplete}/5</span>
   258→      </span>
   259→      <span className="flex items-center gap-1">
   260→        Validation: {complianceData ? <span className="font-medium text-primary">{complianceData.score}% compliant</span> : <span className="font-medium">Not run</span>}
   261→      </span>
   262→      <span className="flex items-center gap-1">
   263→        Approval: <Badge variant="outline" className="h-5 text-[10px] px-1.5 font-mono">{approvalStep}</Badge>
   264→      </span>
   265→      {(transactions.length > 0 || cashbookEntries.length > 0 || ledgerEntries.length > 0) && (
   266→        <span className="flex items-center gap-1">
   267→          Modules: <span className="font-medium text-foreground">{moduleCounts}</span>
   268→        </span>
   269→      )}
   270→    </div>
   271→  )
   272→}

   366→function Step2TrialBalance() {
   367→  const {
   368→    entries, entityConfig, setEntityConfig, updateEntry, deleteEntry, addEntry,
   369→    loadTemplate, saveTrialBalance, isLoading, wizardStep, setWizardStep, autoBalance,
   370→    clearTrialBalance, uploadExcelFile,
   371→  } = useFinancialStore()
   372→  const [autoBalancing, setAutoBalancing] = useState(false)
   373→  const [showAutoBalanceDialog, setShowAutoBalanceDialog] = useState(false)
   374→  const [showClearConfirm, setShowClearConfirm] = useState(false)
   375→  const fileInputRef = React.useRef<HTMLInputElement>(null)
   376→
   377→  const totalDebitCurrent = entries.reduce((s, e) => s + (e.debitCurrent || 0), 0)
   378→  const totalCreditCurrent = entries.reduce((s, e) => s + (e.creditCurrent || 0), 0)
   379→  const totalDebitPrior = entries.reduce((s, e) => s + (e.debitPrior || 0), 0)
   380→  const totalCreditPrior = entries.reduce((s, e) => s + (e.creditPrior || 0), 0)
   381→  const diffCurrent = totalDebitCurrent - totalCreditCurrent
   382→  const diffPrior = totalDebitPrior - totalCreditPrior
   383→  const isBalancedCurrent = Math.abs(diffCurrent) < 0.01
   384→  const isBalancedPrior = Math.abs(diffPrior) < 0.01
   385→  const isBalanced = isBalancedCurrent && isBalancedPrior
   386→
   387→  const handleSave = async () => {
   388→    try {
   389→      await saveTrialBalance()
   390→      toast.success('Trial balance saved and validated')
   391→    } catch {
   392→      toast.error('Failed to save trial balance')
   393→    }
   394→  }
   395→
   396→  const handleAutoBalance = async () => {
   397→    setAutoBalancing(true)
   398→    try {
   399→      await autoBalance()
   400→      toast.success('Trial balance auto-balanced')
   401→      setShowAutoBalanceDialog(false)
   402→    } catch {
   403→      toast.error('Failed to auto-balance')
   404→    } finally {
   405→      setAutoBalancing(false)
   406→    }
   407→  }
   408→
   409→  return (
   410→    <motion.div {...pageVariants} className="py-4 space-y-4">
   411→      {/* Balance Indicator Banner */}
   412→      <Card className={`border-l-4 ${isBalanced ? 'border-l-primary bg-primary/5' : 'border-l-destructive bg-destructive/5'}`}>
   413→        <CardContent className="py-3">
   414→          <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   415→            <div className="flex items-center gap-3">
   416→              {isBalanced
   417→                ? <CheckCircle2 className="h-5 w-5 text-primary" />
   418→                : <AlertCircle className="h-5 w-5 text-destructive" />}
   419→              <div>
   420→                <p className="text-sm font-semibold text-foreground">
   421→                  {isBalanced ? 'Trial Balance is Balanced' : 'Trial Balance is NOT Balanced'}
   422→                </p>
   423→                {!isBalanced && (
   424→                  <p className="text-xs text-muted-foreground mt-0.5">
   425→                    Current diff: {entityConfig.currency} {formatNum(diffCurrent)}
   426→                    {diffPrior !== 0 && ` | Prior diff: ${entityConfig.currency} ${formatNum(diffPrior)}`}
   427→                  </p>
   428→                )}
   429→              </div>
   430→            </div>
   431→            <div className="flex items-center gap-2">
   432→              {!isBalanced && (
   433→                <Dialog open={showAutoBalanceDialog} onOpenChange={setShowAutoBalanceDialog}>
   434→                  <DialogTrigger asChild>
   435→                    <Button variant="outline" size="sm" className="gap-1.5" disabled={entries.length === 0}>
   436→                      <Wand2 className="h-4 w-4" /> Auto-Balance
   437→                    </Button>
   438→                  </DialogTrigger>
   439→                  <DialogContent>
   440→                    <DialogHeader>
   441→                      <DialogTitle>Auto-Balance Trial Balance</DialogTitle>
   442→                      <DialogDescription>
   443→                        This will create a &quot;Suspense&quot; entry to zero out the difference between debits and credits.
   444→                        Current difference: {entityConfig.currency} {formatNum(diffCurrent)}
   445→                      </DialogDescription>
   446→                    </DialogHeader>
   447→                    <DialogFooter>
   448→                      <Button variant="outline" onClick={() => setShowAutoBalanceDialog(false)}>Cancel</Button>
   449→                      <Button onClick={handleAutoBalance} disabled={autoBalancing} className="bg-primary text-primary-foreground gap-2">
   450→                        {autoBalancing ? <Loader2 className="h-4 w-4 animate-spin" /> : <Wand2 className="h-4 w-4" />}
