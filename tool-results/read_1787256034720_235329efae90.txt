   796→function Step4Review() {

   798→  const [loadingCompliance, setLoadingCompliance] = useState(false)
   799→  const [loadingAudit, setLoadingAudit] = useState(false)
   800→  const [autoBalancing, setAutoBalancing] = useState(false)
   801→
   802→  const totalDr = entries.reduce((s, e) => s + (e.debitCurrent || 0), 0)
   803→  const totalCr = entries.reduce((s, e) => s + (e.creditCurrent || 0), 0)
   804→  const tbBalanced = Math.abs(totalDr - totalCr) < 0.01
   805→  const allMapped = entries.every(e => e.category && e.category.trim() !== '')
   806→  const hasBudget = entries.some(e => e.budgetInitial > 0 || e.budgetAdjusted > 0)

   813→      await fetchCompliance()

   815→    } catch {
   816→      toast.error('Failed to run compliance check')
   817→    } finally {
   818→      setLoadingCompliance(false)
   819→    }
   820→  }

   827→      // silent
   828→    } finally {
   829→      setLoadingAudit(false)
   830→    }
   831→  }

   839→      toast.error('Failed to auto-balance')

   847→  }, [activeReportId])

   849→  const validationItems = [
   850→    { label: 'Trial Balance balanced', pass: tbBalanced },
   851→    { label: 'All categories mapped', pass: allMapped },
   852→    { label: 'Budget data provided', pass: hasBudget },
   853→    { label: 'Prior year data present', pass: hasPrior },
   854→    { label: `Entries count: ${entries.length}`, pass: entries.length > 0 },
   855→  ]

   857→  const complianceScore = complianceData?.score ?? 0
   858→
   859→  return (

   885→                {item.pass
